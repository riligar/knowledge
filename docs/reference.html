API Reference

Overview

This documentation provides a complete API reference for Knowledge, including TypeScript interfaces, configurations, public methods, and usage examples.

Main Configuration

DocForgeConfig

Main system configuration interface.

interface DocForgeConfig {
    // Directories
    inputDir: string
    outputDir: string
    templatesDir?: string
    themesDir?: string

    // Site information
    site: SiteConfig

    // Theme and layout
    theme: string
    layout?: string

    // Navigation
    navigation: NavigationConfig

    // Features
    features: FeatureConfig

    // Markdown processing
    markdown: MarkdownConfig

    // Development server
    dev: DevConfig
}

SiteConfig

Basic site configurations.

interface SiteConfig {
    title: string
    description: string
    baseUrl: string
    author: string
    
    // Optional
    keywords?: string[]
    language?: string
    favicon?: string
    logo?: string
    
    // SEO and Social Media
    ogImage?: string
    ogType?: string
    twitterCard?: &#39;summary&#39; | &#39;summary_large_image&#39;
    twitterSite?: string
    
    // Analytics
    googleAnalytics?: string
    gtag?: string
    plausible?: string
    
    // Custom metadata
    meta?: Record&lt;string, string&gt;
}

Example:

site: {
    title: &#39;My Documentation&#39;,
    description: &#39;Complete technical documentation for the project&#39;,
    baseUrl: &#39;/&#39;,
    author: &#39;Development Team&#39;,
    keywords: [&#39;documentation&#39;, &#39;api&#39;, &#39;guides&#39;],
    language: &#39;en-US&#39;,
    favicon: &#39;/assets/favicon.ico&#39;,
    ogImage: &#39;/assets/og-image.png&#39;,
    googleAnalytics: &#39;GA_MEASUREMENT_ID&#39;
}

NavigationConfig

Site navigation configuration.

interface NavigationConfig {
    auto: boolean
    items?: NavigationItem[]
    maxDepth?: number
    showHome?: boolean
    homeTitle?: string
}

interface NavigationItem {
    title: string
    url?: string
    icon?: string
    external?: boolean
    children?: NavigationItem[]
    order?: number
    hidden?: boolean
}

Automatic Navigation:

navigation: {
    auto: true,
    maxDepth: 3,
    showHome: true,
    homeTitle: &#39;Home&#39;
}

Manual Navigation:

navigation: {
    auto: false,
    items: [
        {
            title: &#39;Home&#39;,
            url: &#39;/&#39;,
            icon: &#39;home&#39;
        },
        {
            title: &#39;Guides&#39;,
            children: [
                { title: &#39;Installation&#39;, url: &#39;/guides/installation.html&#39; },
                { title: &#39;Configuration&#39;, url: &#39;/guides/configuration.html&#39; }
            ]
        },
        {
            title: &#39;GitHub&#39;,
            url: &#39;https://github.com/riligar/knowledge&#39;,
            external: true,
            icon: &#39;github&#39;
        }
    ]
}

FeatureConfig

System features configuration.

interface FeatureConfig {
    search?: boolean | SearchConfig
    syntaxHighlight?: boolean | SyntaxHighlightConfig
    darkMode?: boolean | DarkModeConfig
    tableOfContents?: boolean | TableOfContentsConfig
    breadcrumbs?: boolean | BreadcrumbsConfig
    editOnGithub?: string | EditOnGithubConfig
    comments?: boolean | CommentsConfig
    analytics?: boolean | AnalyticsConfig
}

SearchConfig

interface SearchConfig {
    enabled: boolean
    placeholder?: string
    noResultsText?: string
    maxResults?: number
    minQueryLength?: number
    indexFields?: {
        title?: { boost: number }
        excerpt?: { boost: number }
        content?: { boost: number }
        tags?: { boost: number }
    }
    stopWords?: string[]
    stemming?: boolean
}

Example:

search: {
    enabled: true,
    placeholder: &#39;Search documentation...&#39;,
    noResultsText: &#39;No results found&#39;,
    maxResults: 10,
    minQueryLength: 2,
    indexFields: {
        title: { boost: 10 },
        excerpt: { boost: 5 },
        content: { boost: 1 },
        tags: { boost: 8 }
    }
}
